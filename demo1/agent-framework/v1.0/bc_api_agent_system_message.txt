```python
from dotenv import load_dotenv
import os
import requests

# Step 1: Load the .env file into the environment
load_dotenv()  # This loads environment variables from .env file

# Step 2: Fetch credentials from environment variables
client_id = os.getenv('BC_CLIENT_ID')
client_secret = os.getenv('BC_CLIENT_SECRET')
tenant_id = os.getenv('BC_TENANT_ID')
company_id = os.getenv('BC_COMPANY_ID')
environment = os.getenv('BC_ENVIRONMENT', 'sandbox')  # default to sandbox if not set

# Check if environment variables are loaded correctly
if not all([client_id, client_secret, tenant_id, company_id]):
    raise Exception("One or more environment variables are missing.")

# Step 3: Set up OAuth 2.0 Token Endpoint
token_url = f"https://login.microsoftonline.com/{tenant_id}/oauth2/v2.0/token"

# Step 4: Prepare the payload for token request
payload = {
    'grant_type': 'client_credentials',
    'client_id': client_id,
    'client_secret': client_secret,
    'scope': 'https://api.businesscentral.dynamics.com/.default'
}

# Step 5: Request the OAuth 2.0 token
response = requests.post(token_url, data=payload)

# Step 6: Check response and get access token
if response.status_code == 200:
    access_token = response.json().get('access_token')
    print("Access Token successfully acquired")
else:
    raise Exception(f"Failed to get OAuth token: {response.status_code}, {response.text}")

# Step 7: Call the Business Central API with the token
bc_api_url = f"https://api.businesscentral.dynamics.com/v2.0/{tenant_id}/{environment}/api/v2.0/companies({company_id})/financialReports"
headers = {
    "Authorization": f"Bearer {access_token}",
    "Content-Type": "application/json"
}

# Step 8: Make the API request
response = requests.get(bc_api_url, headers=headers)

# Step 9: Output the API response
if response.status_code == 200:
    print("Financial Report Data:", response.json())
else:
    print(f"Error: {response.status_code}, {response.text}")
```
